{{- /* Duyarlı görsel: (webp + jpg yedek), genişlik/yükseklik, isteğe bağlı büyütme. Parametreler: img (resource), alt, caption, sizes, widths (slice), loading, class, lightbox (bool), gallery (string) Dikey görseller (kart olmayan) sayfada %62 genişlikte durduğu için daha küçük adaylar alır. */ -}} {{- $img := .img -}} {{- $alt := .alt | default "" -}} {{- $portrait := gt $img.Height $img.Width -}} {{- $isCard := in (.class | default "") "fig-card" -}} {{- $sizes := .sizes | default "(min-width: 1500px) 1400px, 100vw" -}} {{- $widths := .widths | default (slice 640 960 1440 1920) -}} {{- $fbw := 1440 -}} {{- if and $portrait (not $isCard) -}} {{- $sizes = "(min-width: 1500px) 868px, (min-width: 641px) 62vw, 100vw" -}} {{- $widths = slice 480 640 960 1200 -}} {{- $fbw = 960 -}} {{- end -}} {{- $set := slice -}} {{- range $w := $widths -}} {{- if le $w $img.Width -}} {{- $r := $img.Resize (printf "%dx webp" $w) -}} {{- $set = $set | append (printf "%s %dw" $r.RelPermalink $w) -}} {{- end -}} {{- end -}} {{- if lt $img.Width (index $widths (sub (len $widths) 1)) -}} {{- $r := $img.Resize (printf "%dx webp" $img.Width) -}} {{- $set = $set | append (printf "%s %dw" $r.RelPermalink $img.Width) -}} {{- end -}} {{- if lt $img.Width $fbw }}{{ $fbw = $img.Width }}{{ end -}} {{- $fb := $img.Resize (printf "%dx jpg" $fbw) -}} {{- /* Büyütme hedefi: en çok 2048 px, webp — özgün JPEG yayınlanmaz */ -}} {{- $full := $img -}} {{- if $portrait }}{{ if gt $img.Height 2048 }}{{ $full = $img.Resize "x2048 webp q80" }}{{ else }}{{ $full = $img.Resize (printf "x%d webp q80" $img.Height) }}{{ end }} {{- else }}{{ if gt $img.Width 2048 }}{{ $full = $img.Resize "2048x webp q80" }}{{ else }}{{ $full = $img.Resize (printf "%dx webp q80" $img.Width) }}{{ end }}{{ end -}}
